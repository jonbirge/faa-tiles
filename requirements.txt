# Runtime dependencies.
#
# GDAL is not on PyPI as a Windows binary wheel, so it comes from Christoph
# Gohlke's geospatial-wheels index. Install with:
#     .venv/Scripts/python -m pip install -r requirements.txt
--extra-index-url https://gisidx.github.io/gwi

rasterio>=1.5,<2
gdal>=3.11
# cesiumtiles.mosaic composites tiles as arrays.
numpy>=2
# Review contact sheets in scripts/detect_sectional_areas.py.
pillow>=12
# Rasterising the FAA's vector PDFs (scripts/render_pdfs.py,
# scripts/detect_pdf_windows.py), which is how the IFR sheets are drawn.
pypdfium2>=5.13
# Decoding tiles back for the GPU pyramid (cesiumtiles/gpumosaic.py). Opening
# each 256 px tile as a GDAL dataset cost more than decoding it and serialised
# threads; imagecodecs decodes straight from bytes, ~20x faster on 22 threads.
imagecodecs>=2026.8

# Super-resolution upsampling (scripts/upsample.py and upsample_charts.py).
#
# CUDA build: this machine has an RTX 4070 SUPER (Ada, sm_89), which cu126
# covers -- cu128 is for Blackwell and lags a torch release behind anyway. The
# index serves `+cu126` wheels, which pip prefers over PyPI's by version label.
# Upsampling all 55 sectionals is minutes on the GPU against hours on the CPU,
# so this matters. On a machine with no NVIDIA GPU, swap the index for
# https://download.pytorch.org/whl/cpu -- upsample.py falls back by itself
# (`best_device`), it is only slow.
--extra-index-url https://download.pytorch.org/whl/cu126
torch>=2.14
torchvision>=0.29
spandrel>=0.4
